## Best Practices

### ✅ Do's

```typescript
// ✅ Good: Memoize expensive components
export const MyComponent = React.memo(function MyComponent(props) {
  return <div>{props.content}</div>;
});

// ✅ Good: Use TypeScript for type safety
interface ButtonProps {
  variant: 'primary' | 'secondary';
  onClick: () => void;
  children: React.ReactNode;
}

// ✅ Good: Handle errors gracefully
try {
  // component logic
} catch (error) {
  return <ErrorFallback error={error} />;
}

// ✅ Good: Document complex logic
/**
 * Handles user authentication
 * @param credentials - User credentials
 * @returns Promise with auth token
 */
async function authenticate(credentials) {
  // implementation
}
```

### ❌ Don'ts

```typescript
// ❌ Bad: Inline complex logic in render
return (
  <div>
    {items.map((item, index) => (
      <button
        onClick={() => {
          // complex logic here
          setSomeState(calculateComplexValue(item, index));
        }}
      >
        {item.name}
      </button>
    ))}
  </div>
);

// ❌ Bad: Any types
function processData(data: any) {
  return data.something;
}

// ❌ Bad: Console logs in production
console.log('RELEASE: Component rendered');
```
